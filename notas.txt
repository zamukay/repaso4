public class EstudianteDAO {
    Conexion cc= new Conexion();
    Connection con;
    ResultSet rs;
    PreparedStatement ps ;
    
    public boolean  insertar  (Estudiantes e){
        try {
            String sql="INSERT INTO prueb_3 (cedula,nombre,apellido, nota) VALUES (?,?,?,?)";
            con =cc.conexion();
            ps = con.prepareStatement(sql);
            
            ps.setString(1, e.getCedula());
            ps.setString(2, e.getNombre());
            ps.setString(3, e.getApellido());
            ps.setDouble(4, e.getNota());
            
            ps.executeUpdate();
            
        } catch (SQLException ex) {
            Logger.getLogger(EstudianteDAO.class.getName()).log(Level.SEVERE, null, ex);
        }
                    return true;

    }
    
    public List listar(){
        List<Estudiantes> lista = new ArrayList<>();
       String sql = "SELECT * FROM prueb_3";

        try {

            con = cc.conexion();
            ps = con.prepareStatement(sql);
            rs = ps.executeQuery();

            while (rs.next()) {

                Estudiantes e = new Estudiantes();

                e.setCedula(rs.getString("cedula"));
                e.setNombre(rs.getString("nombre"));
                e.setApellido(rs.getString("apellido"));
                e.setNota(rs.getDouble("nota"));

                lista.add(e);
            }

        } catch (Exception ex) {
            System.out.println(ex);
        }

        return lista;
    }
    
    public boolean eliminar(String cedula){
        try {
            String sql = "DELETE FROM prueb_3 where cedula=?";
            con = cc.conexion();
            ps = con.prepareStatement(sql);
            ps.setString(1, cedula);
            ps.executeUpdate();
            return true;
        } catch (SQLException ex) {
            Logger.getLogger(EstudianteDAO.class.getName()).log(Level.SEVERE, null, ex);
        }
        return false;
    }
    
    public boolean actualizar(Estudiantes e){
                  String sql = "UPDATE  prueb_3 SET nombre=?, apellido=?, nota=? WHERE cedula=?";
        try {
            con = cc.conexion();
            ps = con.prepareStatement(sql);
            ps.setString(1, e.getNombre());
            ps.setString(2, e.getApellido());
            ps.setDouble(3, e.getNota());
            ps.setString(4, e.getCedula());
              
            ps.executeUpdate();
            return true;
        } catch (SQLException ex) {
            Logger.getLogger(EstudianteDAO.class.getName()).log(Level.SEVERE, null, ex);
        }
        return false;
    }
       
    
    
    
}

 */
public class Conexion {
    String url= "jdbc:mysql://localhost:3306/prueb_3";
    String nombre = "root";
    String contrasena = "";
    Connection conectar = null;
    
    public Connection conexion (){
        
        try {
            conectar = DriverManager.getConnection(url,nombre,contrasena);
            System.out.println("Conectado");
        } catch (SQLException ex) {
                Logger.getLogger(Conexion.class.getName()).log(Level.SEVERE, null, ex);
        }
        return conectar;
    }
}
