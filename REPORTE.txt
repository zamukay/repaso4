    public String validarDatos(Estudiante estudiante) {
    StringBuilder errores = new StringBuilder();
    
    // Validar cédula
    if (estudiante.getCedula() == null || estudiante.getCedula().trim().isEmpty()) {
        errores.append("- La cédula no puede estar vacía.\n");
    } else if (!estudiante.getCedula().matches("\\d{10}")) {
        errores.append("- La cédula debe tener exactamente 10 dígitos numéricos.\n");
    }
    
    // Validar nombre
    if (estudiante.getNombre() == null || estudiante.getNombre().trim().isEmpty()) {
        errores.append("- El nombre no puede estar vacío.\n");
    } else if (!estudiante.getNombre().matches("^[\\p{L} .'-]{1,15}$")) {
        errores.append("- El nombre solo permite letras y espacios (hasta 15 caracteres).\n");
    }
    
    // Validar apellido
    if (estudiante.getApellido() == null || estudiante.getApellido().trim().isEmpty()) {
        errores.append("- El apellido no puede estar vacío.\n");
    } else if (!estudiante.getApellido().matches("^[\\p{L} .'-]{1,15}$")) {
        errores.append("- El apellido solo permite letras y espacios (hasta 15 caracteres).\n");
    }
    
    // Validar teléfono (ahora permitimos vacío)
    if (estudiante.getTelefono() != null && !estudiante.getTelefono().trim().isEmpty() && 
        !estudiante.getTelefono().matches("\\d{10}")) {
        errores.append("- El teléfono debe tener exactamente 10 dígitos numéricos.\n");
    }
    
    // Validar dirección (si es requerida)
    if (estudiante.getDireccion() == null || estudiante.getDireccion().trim().isEmpty()) {
        errores.append("- La dirección no puede estar vacía.\n");
    } else if (estudiante.getDireccion().length() > 100) {
        errores.append("- La dirección no puede exceder los 100 caracteres.\n");
    }
    
    return errores.length() > 0 ? errores.toString() : null;
}

    public List<Estudiante> obtenerTodos() {
        List<Estudiante> estudiantes = new ArrayList<>();
        Connection conn = null;
        Statement stmt = null;
        ResultSet rs = null;

        try {
            conn = conexion.conectar();
            stmt = conn.createStatement();
            rs = stmt.executeQuery("SELECT * FROM estudiantes");

            while (rs.next()) {
                Estudiante estudiante = new Estudiante();
                estudiante.setCedula(rs.getString("cedula"));
                estudiante.setNombre(rs.getString("nombre"));
                estudiante.setApellido(rs.getString("apellido"));
                estudiante.setDireccion(rs.getString("direccion"));
                estudiante.setTelefono(rs.getString("telefono"));
                estudiantes.add(estudiante);
            }
        } catch (SQLException e) {
            JOptionPane.showMessageDialog(null, "Error al obtener estudiantes: " + e.getMessage());
            e.printStackTrace();
        } finally {
            cerrarRecursos(conn, stmt, rs);
        }

        return estudiantes;
    }

    public DefaultTableModel cargarTablaEstudiantes() {
        DefaultTableModel modelo = null;
        Connection conn = null;
        Statement stmt = null;
        ResultSet rs = null;

        try {
            String[] titulos = {"Cédula", "Nombre", "Apellido", "Dirección", "Teléfono"};
            modelo = new DefaultTableModel(null, titulos);

            conn = conexion.conectar();
            stmt = conn.createStatement();
            rs = stmt.executeQuery("SELECT * FROM estudiantes");

            while (rs.next()) {
                String[] registro = new String[5];
                registro[0] = rs.getString("cedula");
                registro[1] = rs.getString("nombre");
                registro[2] = rs.getString("apellido");
                registro[3] = rs.getString("direccion");
                registro[4] = rs.getString("telefono");
                modelo.addRow(registro);
            }
        } catch (SQLException e) {
            JOptionPane.showMessageDialog(null, "Error al cargar la tabla: " + e.getMessage());
            e.printStackTrace();
        } finally {
            cerrarRecursos(conn, stmt, rs);
        }

        return modelo;
    }

    public boolean insertar(Estudiante estudiante) {
    Connection conn = null;
    PreparedStatement pstmt = null;
    boolean resultado = false;

    // Validar datos antes de insertar
    String mensajeError = validarDatos(estudiante);
    if (mensajeError != null) {
        JOptionPane.showMessageDialog(null, "Insertar bien los datos");
        return false;
    }

    try {
        conn = conexion.conectar();
        pstmt = conn.prepareStatement("INSERT INTO estudiantes VALUES (?, ?, ?, ?, ?)");
        pstmt.setString(1, estudiante.getCedula());
        pstmt.setString(2, estudiante.getNombre());
        pstmt.setString(3, estudiante.getApellido());
        pstmt.setString(4, estudiante.getDireccion());
        
        // Si el teléfono está vacío, usar 10 ceros
        String telefono = estudiante.getTelefono();
        pstmt.setString(5, telefono == null || telefono.trim().isEmpty() ? "0000000000" : telefono);

        int filas = pstmt.executeUpdate();
        resultado = filas > 0;
    } catch (SQLException e) {
        String errorMsg = "Error al insertar: ";
        if (e.getErrorCode() == 1062) {
            errorMsg += "Cédula ya existe.";
        } else {
            errorMsg += e.getMessage();
        }
        JOptionPane.showMessageDialog(null, errorMsg);

    } finally {
        cerrarRecursos(conn, pstmt, null);
    }

    return resultado;
}
    public boolean actualizar(Estudiante estudiante) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        boolean resultado = false;

        // Validar datos antes de actualizar
        String mensajeError = validarDatos(estudiante);
        if (mensajeError != null) {
            JOptionPane.showMessageDialog(null, mensajeError);
            return false;
        }

        try {
            conn = conexion.conectar();
            pstmt = conn.prepareStatement("UPDATE estudiantes SET nombre=?, apellido=?, direccion=?, telefono=? WHERE cedula=?");
            pstmt.setString(1, estudiante.getNombre());
            pstmt.setString(2, estudiante.getApellido());
            pstmt.setString(3, estudiante.getDireccion());
            pstmt.setString(4, estudiante.getTelefono().isEmpty() ? "0000000000" : estudiante.getTelefono());
            pstmt.setString(5, estudiante.getCedula());

            int filas = pstmt.executeUpdate();
            resultado = filas > 0;
        } catch (SQLException e) {
    JOptionPane.showMessageDialog(null, "Error al actualizar: " + e.getMessage());
            e.printStackTrace();
        } finally {
            cerrarRecursos(conn, pstmt, null);
        }

        return resultado;
    }

    public boolean eliminar(String cedula) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        boolean resultado = false;

        try {
            conn = conexion.conectar();
            pstmt = conn.prepareStatement("DELETE FROM estudiantes WHERE cedula=?");
            pstmt.setString(1, cedula);

            int filas = pstmt.executeUpdate();
            resultado = filas > 0;
        } catch (SQLException e) {
            JOptionPane.showMessageDialog(null, "Error al eliminar estudiante: " + e.getMessage());
            e.printStackTrace();
        } finally {
            cerrarRecursos(conn, pstmt, null);
        }

        return resultado;
    }

    public Estudiante buscarPorCedula(String cedula) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;
        Estudiante estudiante = null;

        try {
            conn = conexion.conectar();
            pstmt = conn.prepareStatement("SELECT * FROM estudiantes WHERE cedula=?");
            pstmt.setString(1, cedula);
            rs = pstmt.executeQuery();

            if (rs.next()) {
                estudiante = new Estudiante();
                estudiante.setCedula(rs.getString("cedula"));
                estudiante.setNombre(rs.getString("nombre"));
                estudiante.setApellido(rs.getString("apellido"));
                estudiante.setDireccion(rs.getString("direccion"));
                estudiante.setTelefono(rs.getString("telefono"));
            }
        } catch (SQLException e) {
            JOptionPane.showMessageDialog(null, "Error al buscar estudiante: " + e.getMessage());
            e.printStackTrace();
        } finally {
            cerrarRecursos(conn, pstmt, rs);
        }

        return estudiante;
    }

    private void cerrarRecursos(Connection conn, Statement stmt, ResultSet rs) {
        try {
            if (rs != null) {
                rs.close();
            }
            if (stmt != null) {
                stmt.close();
            }
            if (conn != null) {
                conn.close();
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

       public void configurarSeleccionTabla() {

                       int fila = jtblEstudiantes.getSelectedRow();
                       JtxtCedula.setText(jtblEstudiantes.getValueAt(fila, 0).toString());

                       textoEditar();
                   }
               }
           });
       }
